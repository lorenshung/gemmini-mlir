"""Narrow prospective source-copy edit, not baseline or compiler authority."""

import ast
import hashlib
from pathlib import Path


def _hash(raw):
    return hashlib.sha256(raw).hexdigest()


def _offset(raw, node, *, end=False):
    row = node.end_lineno if end else node.lineno
    col = node.end_col_offset if end else node.col_offset
    return sum(map(len, raw.splitlines(keepends=True)[: row - 1])) + col


def _leaf(root, name):
    path = root / name
    if (
        path.is_symlink()
        or path.resolve(strict=True) != path
        or path.stat().st_nlink != 1
    ):
        raise ValueError("panel preparation needs an original unaliased source copy")
    return path.read_bytes()


def apply_to_copy(package):
    """Apply after optional transpose-pack entry, before final Builder cleanup.

    The original preparation/edit authority must validate the raw baseline and
    own this copied root. No whole contraction file or existing helper is
    replaced; source changes are strict original lexical/AST positions only.
    """
    root = Path(package)
    if not root.is_absolute() or root.resolve(strict=True) != root or root.is_symlink():
        raise ValueError("panel preparation requires its canonical copied package")
    if (root / "rhs_panel.py").exists():
        raise ValueError("panel helper already exists; preparation refuses overwrite")
    originals = {n: _leaf(root, n) for n in ("backend.py", "contraction.py")}
    raw = originals["backend.py"]
    tree = ast.parse(raw)
    calls = [
        n
        for n in ast.walk(tree)
        if isinstance(n, ast.Call)
        and isinstance(n.func, ast.Name)
        and n.func.id == "matmul"
    ]
    if len(calls) != 1 or any(k.arg == "_panel_owned" for k in calls[0].keywords):
        raise ValueError("panel preparation has no unique original backend matmul call")
    if not any(
        isinstance(n, ast.Name)
        and n.id == "cached_handle"
        and isinstance(n.ctx, ast.Store)
        for n in ast.walk(tree)
    ):
        raise ValueError("panel preparation lacks original cached-RHS lifetime owner")
    point = _offset(raw, calls[0], end=True) - 1
    if raw[point : point + 1] != b")":
        raise ValueError("panel backend call lexical closure differs")
    backend = raw[:point] + b",_panel_owned=cached_handle is None" + raw[point:]
    raw = originals["contraction.py"]
    tree = ast.parse(raw)
    functions = [
        n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == "matmul"
    ]
    if len(functions) != 1:
        raise ValueError("panel preparation needs one original matmul definition")
    function = functions[0]
    if (
        function.args.vararg
        or function.args.kwarg
        or not function.args.defaults
        or any(a.arg == "_panel_owned" for a in function.args.args)
    ):
        raise ValueError("panel signature has changed or already contains the hook")
    anchors = [
        n
        for n in function.body
        if isinstance(n, ast.Assign)
        and len(n.targets) == 1
        and isinstance(n.targets[0], ast.Name)
        and n.targets[0].id == "epilogue"
        and isinstance(n.value, ast.Call)
        and isinstance(n.value.func, ast.Attribute)
        and isinstance(n.value.func.value, ast.Name)
        and n.value.func.value.id == "attrs"
        and n.value.func.attr == "get"
        and len(n.value.args) == 2
        and isinstance(n.value.args[0], ast.Constant)
        and n.value.args[0].value == "epilogue"
        and isinstance(n.value.args[1], ast.List)
        and not n.value.args[1].elts
    ]
    if len(anchors) != 1:
        raise ValueError(
            "panel preparation needs the unique original epilogue entry boundary"
        )
    if any(
        isinstance(n, ast.ImportFrom) and n.module == "rhs_panel"
        for n in ast.walk(function)
    ):
        raise ValueError("panel entry hook already exists")
    # Earlier A pack entry stays before this original boundary. All full/tail
    # source computations and later B Builder cleanup remain separately owned.
    entry = b"""    from rhs_panel import try_emit
    if try_emit(b, a, w, out, m, k, n, attrs,
                private_rows_owned=_panel_owned, transpose_rhs=transpose_rhs,
                resident=_resident):
        return
"""
    boundary = _offset(raw, anchors[0]) - anchors[0].col_offset
    signature = _offset(raw, function.args.defaults[-1], end=True)
    contraction = raw[:boundary] + entry + raw[boundary:]
    contraction = (
        contraction[:signature] + b", _panel_owned=False" + contraction[signature:]
    )
    ast.parse(backend)
    ast.parse(contraction)
    helper = Path(__file__).with_name("gemmini_v26_rhs_panel.py").read_bytes()
    ast.parse(helper)
    for name, old in originals.items():
        if _leaf(root, name) != old:
            raise ValueError("panel source copy changed before mutation")
    for name, new in [
        ("backend.py", backend),
        ("contraction.py", contraction),
        ("rhs_panel.py", helper),
    ]:
        (root / name).write_bytes(new)
    return {
        "schema": "gemmini.v26.rhs_panel_preparation.v1",
        "before": {n: _hash(v) for n, v in originals.items()},
        "after": {
            n: _hash((root / n).read_bytes())
            for n in ["backend.py", "contraction.py", "rhs_panel.py"]
        },
        "order": "optional pack-entry, panel-entry, post-schedule configuration/fence cleanup",
        "execution_qualification": False,
        "timing_authority": "NONE",
    }
